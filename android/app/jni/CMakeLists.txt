cmake_minimum_required(VERSION 3.22)

project(TwistedMetal4Android LANGUAGES C CXX)

# Gradle invokes this file from android/app. The game project remains the
# authoritative CMake root so desktop and Android share the same source graph.
set(TM4_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH
    "Twisted Metal 4 Recomp project root" FORCE)
set(PSXRECOMP_ROOT "${TM4_ROOT}/psxrecomp" CACHE PATH
    "Path to psxrecomp" FORCE)

# The root CMakeLists.txt turns psx-runtime into libmain.so on Android and
# disables desktop-only surfaces. Use a separate binary directory to avoid
# colliding with a host build generated by the setup tools.
add_subdirectory("${TM4_ROOT}" "${CMAKE_BINARY_DIR}/twistedmetal4")
